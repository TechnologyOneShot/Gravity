cmake_minimum_required(VERSION 3.16)
project(gravity_image_check C CXX)
# Builds Gravity's picture engine core (engine-image) with stable-diffusion.cpp at SD_DIR, for CI.
set(SD_DIR "" CACHE PATH "stable-diffusion.cpp checkout (with submodules)")
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${SD_DIR} sd-build)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../../engine-image/src/main/cpp)
add_executable(gravity-image-cli gravity_image_cli.cpp ${CORE}/gravity_image_core.cpp)
target_include_directories(gravity-image-cli PRIVATE ${CORE} ${SD_DIR}/include)
target_compile_features(gravity-image-cli PRIVATE cxx_std_17)
target_link_libraries(gravity-image-cli stable-diffusion)
